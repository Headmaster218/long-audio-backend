        fragment_cleanup_error=None,initializer_safe=True);values.update(changes)

def test_original_conjunction_branches_and_cleanup_calls_are_unchanged():
    old=ast.parse(BASE.read_text());new=ast.parse(Path(app_module.__file__).read_text())
    expected=ast.dump(ast.parse(PREDICATE,mode='eval').body,include_attributes=False)
    def decision(tree):
        found=[n for n in ast.walk(tree) if isinstance(n,ast.If)
            and ast.dump(n.test,include_attributes=False)==expected]
        assert len(found)==1
        return ast.dump(found[0],include_attributes=False)
    assert decision(old)==decision(new)
    names={'stop','close','close_admission','unregister_cells','revoke_recovery','release','health'}
    def calls(tree):
        return sorted(ast.dump(n,include_attributes=False) for n in ast.walk(tree) if isinstance(n,ast.Call)
            and ((isinstance(n.func,ast.Attribute) and n.func.attr in names)
                or isinstance(n.func,ast.Name) and n.func.id in names))
    assert calls(old)==calls(new)


@pytest.mark.parametrize('stopped',[True,False])
@pytest.mark.parametrize('diagnostic_fails',[True,False])
def test_real_lifespan_keeps_worker_result_and_single_stop(tmp_path,stopped,diagnostic_fails):
    app=app_module.create_app(Settings(data_dir=tmp_path,disk_reserve_bytes=0,start_worker=False),analyzer_factory=FakeAnalyzer)
    worker=app.state.worker;original=worker.stop;calls=[]
    def stop():calls.append(True);assert original();return stopped
    def observe(*a,**k):
        if diagnostic_fails:raise KeyboardInterrupt('synthetic diagnostic failure')
        return saved(*a,**k)
    saved=diagnostics.record_shutdown_decision
    with patch.object(worker,'stop',side_effect=stop),patch.object(diagnostics,'record_shutdown_decision',side_effect=observe):
        if stopped:
            with TestClient(app):pass
        else:
            with pytest.raises(RuntimeError,match='Shutdown blocked'):
                with TestClient(app):pass
    assert calls==[True]
    assert app.state.instance_lock_cleanup_complete is stopped
    if not diagnostic_fails:
        assert app.state.shutdown_decision_diagnostics['predicates']['worker_stopped'] is stopped
    if not stopped:
        # Test-only original stop has already positively completed. The production
        # app correctly retained its lock on the injected false acknowledgment.
        # Close the test's lock only to avoid leaking a fixture-owned descriptor.
        app.state.instance_lock.release()


def test_app_unwinding_exception_keeps_identity_when_diagnostic_fails(tmp_path):
    app=app_module.create_app(Settings(data_dir=tmp_path,disk_reserve_bytes=0,start_worker=False),analyzer_factory=FakeAnalyzer)
    original=RuntimeError('original lifespan body error')
    async def unwind():
        async with app.router.lifespan_context(app):raise original
    with patch.object(diagnostics,'record_shutdown_decision',side_effect=KeyboardInterrupt('diagnostic')):
        with pytest.raises(RuntimeError) as caught:
            asyncio.run(unwind())
    assert caught.value is original
    assert app.state.instance_lock_cleanup_complete is True
